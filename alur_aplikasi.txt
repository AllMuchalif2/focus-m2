TAHAP 2: PEMERIKSAAN STATUS LOGIN (AUTH GATE)
File: lib/features/auth/auth_gate.dart
--------------------------------------------------------------------------------
AuthGate memantau Stream authStateChanges dari Firebase:

class AuthGate extends StatelessWidget {
  const AuthGate({super.key});

  @override
  Widget build(BuildContext context) {
    final authService = context.watch<AuthService>();

    return StreamBuilder<AppUser?>(
      stream: authService.authStateChanges,
      builder: (context, snapshot) {
        if (snapshot.connectionState == ConnectionState.waiting) {
          return const Scaffold(
            body: Center(
              child: CircularProgressIndicator(),
            ),
          );
        }

        final user = snapshot.data;
        if (user == null) {
          return const LoginScreen(); // Jika belum login -> ke LoginScreen
        }

        return const TimerScreen(); // Jika sudah login -> ke TimerScreen
      },
    );
  }
}

// Audio diputar berulang (looping) secara non-blocking:
Future<void> playAmbience(String url) async {
  try {
    if (_currentAmbienceUrl == url && _ambiencePlayer.playing) return;

    _currentAmbienceUrl = url;
    notifyListeners();

    await _ambiencePlayer.setUrl(url);
    await _ambiencePlayer.setLoopMode(LoopMode.one);
    await _ambiencePlayer.setVolume(_ambienceVolume);
    _ambiencePlayer.play();
  } catch (_) {
    if (_currentAmbienceUrl == url) {
      _currentAmbienceUrl = null;
      notifyListeners();
    }
  }
}

// Video vertikal di-mute dan diputar barengan tanpa menghentikan audio:
Future<void> setVideoBackground(String? url) async {
  if (_currentVideoUrl == url && _videoController != null) return;

  if (_videoController != null) {
    await _videoController!.pause();
    await _videoController!.dispose();
    _videoController = null;
    _isVideoInitialized = false;
  }

  _currentVideoUrl = url;
  notifyListeners();

  if (url != null && url.isNotEmpty) {
    try {
      final controller = VideoPlayerController.networkUrl(
        Uri.parse(url),
        videoPlayerOptions: VideoPlayerOptions(mixWithOthers: true),
      );
      _videoController = controller;
      await controller.initialize();
      await controller.setVolume(0.0);
      await controller.setLooping(true);
      await controller.play();
      _isVideoInitialized = true;
    } catch (_) {
      _videoController = null;
      _isVideoInitialized = false;
    }
  }

  notifyListeners();
}

Future<void> scheduleTimerNotification({
  required DateTime endAt,
  required String title,
  required String body,
  int id = 1001,
}) async {
  await initialize();

  final scheduledDate = tz.TZDateTime.from(endAt, tz.local);
  if (scheduledDate.isBefore(tz.TZDateTime.now(tz.local))) {
    return;
  }

  const notificationDetails = NotificationDetails(
    android: AndroidNotificationDetails(
      'timer_channel',
      'Timer Notifications',
      channelDescription: 'Pemberitahuan saat sesi Pomodoro selesai',
      importance: Importance.max,
      priority: Priority.high,
      playSound: true,
    ),
  );

  try {
    await _notificationsPlugin.zonedSchedule(
      id,
      title,
      body,
      scheduledDate,
      notificationDetails,
      androidScheduleMode: AndroidScheduleMode.exactAllowWhileIdle,
      uiLocalNotificationDateInterpretation:
          UILocalNotificationDateInterpretation.absoluteTime,
    );
  } catch (_) {
    try {
      await _notificationsPlugin.zonedSchedule(
        id,
        title,
        body,
        scheduledDate,
        notificationDetails,
        androidScheduleMode: AndroidScheduleMode.inexactAllowWhileIdle,
        uiLocalNotificationDateInterpretation:
            UILocalNotificationDateInterpretation.absoluteTime,
      );
    } catch (_) {}
  }
}

Saat pergantian fase (Fokus -> Istirahat, atau Istirahat -> Fokus), sistem langsung mengirim notifikasi instan dan membunyikan alarm transisi:
File: lib/services/timer_service.dart
Future<void> _onPhaseTransitioned() async {
  final isNowBreak = _engine.phase.isBreak;
  final alertTitle = isNowBreak ? 'Sesi Fokus Selesai!' : 'Waktu Istirahat Selesai!';
  final alertBody = isNowBreak
      ? 'Waktunya istirahat (${_engine.currentPhaseDuration.inMinutes} menit).'
      : 'Mari mulai fokus kembali (${_engine.currentPhaseDuration.inMinutes} menit).';

  await _notificationService.showImmediateNotification(
    title: alertTitle,
    body: alertBody,
    id: 1002,
  );

  _mediaService?.playAlarm();

  if (_engine.endAt != null) {
    await _notificationService.scheduleTimerNotification(
      endAt: _engine.endAt!,
      title: isNowBreak ? 'Istirahat Selesai' : 'Sesi Selesai',
      body: 'Fase ${_engine.phase.displayName} telah berakhir.',
    );
  }
  await _saveState();
  notifyListeners();
}

Fungsi _onWillPop menampilkan dialog konfirmasi:
Future<void> _onWillPop(TimerService timerService, MediaService mediaService) async {
  final isSessionActive = timerService.isRunning || timerService.isPaused;
  if (!isSessionActive) {
    if (Navigator.of(context).canPop()) {
      Navigator.of(context).pop();
    } else {
      await SystemNavigator.pop();
    }
    return;
  }

  final theme = Theme.of(context);
  final result = await showDialog<String>(
    context: context,
    builder: (ctx) => AlertDialog(
      backgroundColor: theme.colorScheme.surface,
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(24)),
      content: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          Container(
            width: 56, height: 56,
            decoration: BoxDecoration(
              color: theme.colorScheme.error.withValues(alpha: 0.15),
              shape: BoxShape.circle,
            ),
            child: Icon(Icons.warning_amber_rounded, color: theme.colorScheme.error, size: 32),
          ),
          const SizedBox(height: 16),
          Text('Sesi Masih Berjalan', style: theme.textTheme.titleLarge?.copyWith(fontWeight: FontWeight.bold)),
          const SizedBox(height: 8),
          Text('Keluar sekarang akan menghentikan timer dan musik yang sedang berjalan.'),
          const SizedBox(height: 24),
          FilledButton.icon(
            onPressed: () => Navigator.pop(ctx, 'stay'),
            icon: const Icon(Icons.play_arrow_rounded),
            label: const Text('Tetap di Sini'),
          ),
          const SizedBox(height: 10),
          OutlinedButton.icon(
            onPressed: () => Navigator.pop(ctx, 'stop'),
            icon: const Icon(Icons.stop_rounded),
            label: const Text('Hentikan Sesi'),
          ),
        ],
      ),
    ),
  );

  if (result == 'stop') {
    await timerService.stop();
    await mediaService.stopAmbience();
  }
}

Future<void> _saveSession() async {
  setState(() => _isSaving = true);

  try {
    final user = context.read<AuthService>().currentUser;
    if (user != null) {
      final session = SessionRecord(
        id: '',
        userId: user.id,
        mode: widget.completedMode.name,
        durationMinutes: widget.completedDuration.inMinutes,
        reflection: _reflectionController.text.trim(),
        hasProof: false,
        createdAt: DateTime.now(),
      );

      // Simpan ke database Firestore
      await context.read<SessionRepository>().createSession(user.id, session);
    }

    final timerService = context.read<TimerService>();
    await timerService.stop();

    if (mounted) {
      Navigator.pop(context);
    }
  } catch (e) {
    // Penanganan error
  }
}

Future<void> _confirmLogout() async {
  final theme = Theme.of(context);
  final confirmed = await showDialog<bool>(
    context: context,

            icon: const Icon(Icons.logout_rounded),

          ),
          const SizedBox(height: 10),
          OutlinedButton(
            onPressed: () => Navigator.pop(ctx, false),
            child: const Text('Batal'),
          ),
        ],
      ),
    ),
  );

  // Jika user menyetujui logout:
  if (confirmed == true && mounted) {
    await context.read<TimerService>().stop(); // Hentikan timer & notifikasi
    if (mounted) {
      await context.read<MediaService>().stopAmbience(); // Hentikan audio & video
    }
    if (mounted) {
      await context.read<AuthService>().logout(); // Firebase sign out
    }
  }
}
