2. Jika koneksi masih memeriksa (waiting), Scaffold menampilkan CircularProgressIndicator di tengah layar.
3. Jika snapshot.data bernilai null (pengguna belum login), AuthGate merender LoginScreen.

Kode Terkait:
File: lib/features/auth/auth_gate.dart
```dart
class AuthGate extends StatelessWidget {
  const AuthGate({super.key});

  @override
  Widget build(BuildContext context) {
    final authService = context.watch<AuthService>();

    return StreamBuilder<AppUser?>(
      stream: authService.authStateChanges,
      builder: (context, snapshot) {
        if (snapshot.connectionState == ConnectionState.waiting) {
          return const Scaffold(
            body: Center(
              child: CircularProgressIndicator(),
            ),
          );
        }

        final user = snapshot.data;
        if (user == null) {
          return const LoginScreen();
        }

        return const TimerScreen();
      },
    );
  }
}
```

  return _firebaseAuth
      .authStateChanges()

Future<AppUser> register({
  required String email,
  required String password,
}) async {
  final credential = await _firebaseAuth.createUserWithEmailAndPassword(
    email: email.trim(),
    password: password,
  );
  final user = credential.user;
  if (user == null) throw 'Gagal membuat pengguna.';
  return AppUser.fromFirebaseUser(user);
}

Future<AppUser> login({
  required String email,
  required String password,
}) async {
  final credential = await _firebaseAuth.signInWithEmailAndPassword(
    email: email.trim(),
    password: password,
  );
  final user = credential.user;
  if (user == null) throw 'Gagal masuk akun.';
  return AppUser.fromFirebaseUser(user);
}

Future<void> playAmbience(String url) async {
  try {
    if (_currentAmbienceUrl == url && _ambiencePlayer.playing) return;
    _currentAmbienceUrl = url;
    notifyListeners();

    await _ambiencePlayer.setUrl(url);
    await _ambiencePlayer.setLoopMode(LoopMode.one);
    await _ambiencePlayer.setVolume(_ambienceVolume);
    _ambiencePlayer.play();
  } catch (_) {
    if (_currentAmbienceUrl == url) {
      _currentAmbienceUrl = null;
      notifyListeners();
    }
  }
}

Future<void> setVideoBackground(String? url) async {
  if (_currentVideoUrl == url && _videoController != null) return;

  if (_videoController != null) {
    await _videoController!.pause();
    await _videoController!.dispose();
    _videoController = null;
    _isVideoInitialized = false;
  }

  _currentVideoUrl = url;
  notifyListeners();

  if (url != null && url.isNotEmpty) {
    try {
      final controller = VideoPlayerController.networkUrl(
        Uri.parse(url),
        videoPlayerOptions: VideoPlayerOptions(mixWithOthers: true),
      );
      _videoController = controller;
      await controller.initialize();
      await controller.setVolume(0.0);
      await controller.setLooping(true);
      await controller.play();
      _isVideoInitialized = true;
    } catch (_) {
      _videoController = null;
      _isVideoInitialized = false;
    }
  }
  notifyListeners();
}

Future<void> _scheduleAllPhaseNotifications() async {
  await _notificationService.cancelAllNotifications();
  if (_engine.startAt == null) return;

  DateTime phaseStart = _engine.startAt!;
  for (int i = 0; i < _engine.plans.length; i++) {
    final plan = _engine.plans[i];
    final phaseEnd = phaseStart.add(plan.duration);
    final isLast = i == _engine.plans.length - 1;

    final title = isLast
        ? 'Sesi Selesai!'
        : (plan.phase.isFocus ? 'Sesi Fokus Selesai!' : 'Waktu Istirahat Selesai!');
    final body = isLast
        ? 'Seluruh sesi fokus telah selesai. Bagus sekali!'
        : (plan.phase.isFocus ? 'Waktunya istirahat.' : 'Mari mulai fokus kembali.');

    await _notificationService.scheduleTimerNotification(
      endAt: phaseEnd,
      title: title,
      body: body,
      id: 1000 + i,
    );
    phaseStart = phaseEnd;
  }
}

Future<void> start() async {
  _engine.start(DateTime.now());
  _startTicker();
  await _updateWakelock(true);
  await _scheduleAllPhaseNotifications();
  await _saveState();
  notifyListeners();
}

Future<void> scheduleTimerNotification({
  required DateTime endAt,
  required String title,
  required String body,
  int id = 1001,
}) async {
  await initialize();
  final scheduledDate = tz.TZDateTime.from(endAt, tz.local);
  if (scheduledDate.isBefore(tz.TZDateTime.now(tz.local))) return;

  const notificationDetails = NotificationDetails(
    android: AndroidNotificationDetails(
      'timer_channel',
      'Timer Notifications',
      channelDescription: 'Pemberitahuan saat sesi Pomodoro selesai',
      importance: Importance.max,
      priority: Priority.high,
      playSound: true,
    ),
  );

  try {
    await _notificationsPlugin.zonedSchedule(
      id,
      title,
      body,
      scheduledDate,
      notificationDetails,
      androidScheduleMode: AndroidScheduleMode.exactAllowWhileIdle,
      uiLocalNotificationDateInterpretation:
          UILocalNotificationDateInterpretation.absoluteTime,
    );
  } catch (_) {
    try {
      await _notificationsPlugin.zonedSchedule(
        id,
        title,
        body,
        scheduledDate,
        notificationDetails,
        androidScheduleMode: AndroidScheduleMode.inexactAllowWhileIdle,
        uiLocalNotificationDateInterpretation:
            UILocalNotificationDateInterpretation.absoluteTime,
      );
    } catch (_) {}
  }
}

TAHAP 7: PROTEKSI TOMBOL BACK SAAT SESI BERJALAN
--------------------------------------------------------------------------------
Komponen yang Terlibat:

- Dialog Widget   : AlertDialog dengan 3 aksi tombol
- Services Terkait: TimerService (stop), MediaService (stopAmbience)

Penjelasan Alur:
1. Layar TimerScreen dibungkus PopScope(canPop: false).
2. Jika tombol Back HP ditekan saat sesi idle/selesai: aplikasi keluar secara normal via Navigator.pop() atau SystemNavigator.pop().
3. Jika tombol Back ditekan saat sesi SEDANG BERJALAN (running atau paused), sistem memunculkan AlertDialog dengan 3 opsi:
   - 'Tetap di Sini' : Menutup dialog; sesi timer dan audio tetap berjalan tanpa gangguan.
   - 'Minimize'      : Mengeksekusi SystemNavigator.pop(); aplikasi pindah ke latar belakang (background) seperti menekan tombol Home, sehingga timer dan musik tetap aktif.
   - 'Hentikan Sesi' : Mematikan timer (stop()), menghentikan musik/video (stopAmbience()), lalu keluar dari aplikasi.

Kode Terkait:
File: lib/features/timer/timer_screen.dart
```dart
return PopScope(
  canPop: false,
  onPopInvokedWithResult: (didPop, _) async {
    if (didPop) return;
    await _onWillPop(timerService, mediaService);
  },
  child: Scaffold(...),
);

Future<void> _onWillPop(TimerService timerService, MediaService mediaService) async {
  final isSessionActive = timerService.isRunning || timerService.isPaused;
  if (!isSessionActive) {
    if (Navigator.of(context).canPop()) {
      Navigator.of(context).pop();
    } else {
      await SystemNavigator.pop();
    }
    return;
  }

  final theme = Theme.of(context);
  final result = await showDialog<String>(
    context: context,
    builder: (ctx) => AlertDialog(
      title: const Text('Sesi Masih Berjalan'),
      content: const Text('Keluar sekarang akan menghentikan timer dan musik yang sedang berjalan.'),
      actions: [
        FilledButton.icon(
          onPressed: () => Navigator.pop(ctx, 'stay'),
          icon: const Icon(Icons.play_arrow_rounded),
          label: const Text('Tetap di Sini'),
        ),
        OutlinedButton.icon(
          onPressed: () => Navigator.pop(ctx, 'minimize'),
          icon: const Icon(Icons.minimize_rounded),
          label: const Text('Minimize'),
        ),
        OutlinedButton.icon(
          onPressed: () => Navigator.pop(ctx, 'stop'),
          icon: const Icon(Icons.stop_rounded),
          label: const Text('Hentikan Sesi'),
        ),
      ],
    ),
  );

  if (result == 'minimize') {
    await SystemNavigator.pop();
  } else if (result == 'stop') {
    await timerService.stop();
    await mediaService.stopAmbience();
    if (mounted) {
      if (Navigator.of(context).canPop()) {
        Navigator.of(context).pop();
      } else {
        await SystemNavigator.pop();
      }
    }
  }
}
```

--------------------------------------------------------------------------------
Komponen yang Terlibat:
- Screens / Widgets:
  * lib/features/sessions/sessions_history_screen.dart (SessionsHistoryScreen)
  * lib/features/sessions/session_detail_screen.dart (SessionDetailScreen)
  * lib/features/stats/stats_screen.dart (StatsScreen)
  * lib/features/stats/stats_calculator.dart (StatsCalculator)
  * lib/features/stats/widgets/bar_chart_widget.dart (BarChartWidget)
- Data Layer       :
  * lib/data/repositories/session_repository.dart
  * lib/data/models/session_record.dart (SessionRecord)
- Service Terkait  : lib/services/auth_service.dart

Penjelasan Alur:
1. Tombol Riwayat di AppBar membuka SessionsHistoryScreen:
   - Menggunakan StreamBuilder yang memanggil SessionRepository.watchRecentSessions() untuk menampilkan 10 sesi terakhir secara real-time.
   - Pengguna dapat mengetuk sesi untuk membuka SessionDetailScreen (melihat catatan refleksi lengkap atau menghapus sesi).
2. Tombol Statistik di AppBar membuka StatsScreen:
   - Mengambil data sesi 30 hari terakhir via SessionRepository.getSessionsForDateRange().
   - StatsCalculator menghitung total menit fokus hari ini, minggu ini, dan rekor streak hari berturut-turut.
   - BarChartWidget menggambar grafik batang native tanpa library pihak ketiga.

Kode Terkait:
File: lib/features/sessions/sessions_history_screen.dart
```dart
StreamBuilder<List<SessionRecord>>(
  stream: sessionRepo.watchRecentSessions(user.id, limit: 10),
  builder: (context, snapshot) {
    if (snapshot.connectionState == ConnectionState.waiting) {
      return const Center(child: CircularProgressIndicator());
    }
    final sessions = snapshot.data ?? [];
    return ListView.builder(
      itemCount: sessions.length,
      itemBuilder: (context, index) {
        final session = sessions[index];
        return ListTile(
          title: Text('${session.durationMinutes} Menit Fokus'),
          subtitle: Text(session.reflection),
          onTap: () => Navigator.push(
            context,
            MaterialPageRoute(
              builder: (_) => SessionDetailScreen(session: session),
            ),
          ),
        );
      },
    );
  },
)
```

File: lib/features/stats/stats_screen.dart
```dart
void _loadStats() {
  final user = context.read<AuthService>().currentUser;
  if (user != null) {
    final now = DateTime.now();
    final startDate = now.subtract(const Duration(days: 30));
    _statsFuture = context.read<SessionRepository>().getSessionsForDateRange(
      user.id,
      start: startDate,
      end: now,
    );
  }
}
```

TAHAP 10: KELUAR AKUN (LOGOUT)
--------------------------------------------------------------------------------
Komponen yang Terlibat:
- Screen / Widget : lib/features/timer/timer_screen.dart (_confirmLogout)
- Dialog Widget   : AlertDialog konfirmasi logout
- Services Terkait:
  * lib/services/timer_service.dart (stop())
  * lib/services/media_service.dart (stopAmbience())
  * lib/services/auth_service.dart (logout())
- Navigation Gate : lib/features/auth/auth_gate.dart (AuthGate)

Penjelasan Alur:
1. Pengguna menekan tombol icon logout di AppBar TimerScreen.
2. Dialog konfirmasi muncul menanyakan kepastian keluar akun.
3. Jika disetujui:
   - timerService.stop() membatalkan semua alarm notifikasi dan mereset status timer ke awal.
   - mediaService.stopAmbience() mematikan pemutaran audio dan merilis controller video.
   - authService.logout() mengeksekusi signOut() pada FirebaseAuth.
4. Stream authStateChanges pada AuthGate mendeteksi perubahan nilai user menjadi null, sehingga aplikasi otomatis kembali menampilkan LoginScreen.

Kode Terkait:
File: lib/features/timer/timer_screen.dart
```dart
Future<void> _confirmLogout() async {
  final confirmed = await showDialog<bool>(
    context: context,
    builder: (ctx) => AlertDialog(
      title: const Text('Keluar Akun?'),
      content: const Text('Kamu akan keluar dari akun ini. Sesi yang sedang berjalan akan dihentikan.'),
      actions: [
        FilledButton.icon(
          onPressed: () => Navigator.pop(ctx, true),
          icon: const Icon(Icons.logout_rounded),
          label: const Text('Keluar'),
        ),
        OutlinedButton(
          onPressed: () => Navigator.pop(ctx, false),
          child: const Text('Batal'),
        ),
      ],
    ),
  );

  if (confirmed == true && mounted) {
    await context.read<TimerService>().stop();
    if (mounted) {
      await context.read<MediaService>().stopAmbience();
    }
    if (mounted) {
      await context.read<AuthService>().logout();
    }
  }
}
```

File: lib/services/auth_service.dart
```dart
Future<void> logout() async {
  try {
    await _firebaseAuth.signOut();
  } on FirebaseAuthException catch (e) {
    throw _handleFirebaseAuthException(e);
  } catch (e) {
    throw e.toString();
  }
}
```
